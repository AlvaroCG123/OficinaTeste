// schema.prisma

generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "mysql"
  url      = env("DATABASE_URL")
}

// ---------------------------------------------------------
// 1. A REDE: UNIDADES E COLABORADORES
// ---------------------------------------------------------

model Unidade {
  id             Int       @id @default(autoincrement())
  nome           String
  endereco       String
  telefone       String
  ativa          Boolean   @default(true)
  
  // RN31: Unidade pode ficar sem gerente (Int?). Isso desata o nó 
  // para permitir o primeiro insert com banco vazio.
  gerenteId      Int?      @unique 
  gerente        Colaborador? @relation("GerenteUnidade", fields: [gerenteId], references: [id])
  
  colaboradores  Colaborador[] @relation("LotacaoUnidade")
  ordensServico  OrdemServico[]
}

model Colaborador {
  id               Int       @id @default(autoincrement())
  nome             String
  email            String    @unique // RN03
  cpf              String    @unique
  senha            String
  perfil           Perfil
  dataAdmissao     DateTime  @default(now())
  ativo            Boolean   @default(true)

  // O colaborador OBRIGATORIAMENTE pertence a uma unidade
  unidadeId        Int
  unidade          Unidade   @relation("LotacaoUnidade", fields: [unidadeId], references: [id])
  
  // Relação reversa de Gerente
  unidadeGerenciada Unidade? @relation("GerenteUnidade")

  // AUTO-RELACIONAMENTO: Supervisor (Chefe)
  supervisorId     Int?
  supervisor       Colaborador?  @relation("HierarquiaColaborador", fields: [supervisorId], references: [id])
  supervisionados  Colaborador[] @relation("HierarquiaColaborador")

  // Relações com as operações do dia a dia
  osAtendidas      OrdemServico[] @relation("AtendenteOS")
  osExecutadas     OrdemServico[] @relation("MecanicoOS")
  servicosFeitos   ItemOSServico[]
  pagamentosRec    Pagamento[]
  movimentacoes    MovimentacaoEstoque[]
}

enum Perfil {
  ADMIN
  DIRETORIA
  GERENTE
  CHEFE_DE_OFICINA
  ATENDIMENTO
  MECANICO
}

// ---------------------------------------------------------
// 2. CLIENTES, TELEFONES E VEÍCULOS
// ---------------------------------------------------------

model Cliente {
  id             Int       @id @default(autoincrement())
  nome           String    // Nome completo ou Razão Social
  documento      String    @unique // CPF ou CNPJ (RN02)
  tipoPessoa     TipoPessoa
  email          String?
  dataNascimento DateTime? // Só preenchido se for PF
  
  // Endereço unificado na própria tabela pra simplificar
  logradouro     String
  numero         String
  complemento    String?
  bairro         String
  cidade         String
  uf             String
  cep            String
  observacoes    String?   @db.Text

  telefones      Telefone[]
  veiculos       Veiculo[]
}

enum TipoPessoa {
  FISICA
  JURIDICA
}

model Telefone {
  id        Int     @id @default(autoincrement())
  numero    String
  clienteId Int
  // RN17: Apagar o cliente apaga o telefone (Cascade)
  cliente   Cliente @relation(fields: [clienteId], references: [id], onDelete: Cascade)

  // RN06: O mesmo número não entra duas vezes para o MESMO cliente
  @@unique([clienteId, numero])
}

model Veiculo {
  id             Int       @id @default(autoincrement())
  placa          String    @unique // RN01
  marca          String
  modelo         String
  anoFabricacao  Int       // RN29: Int é mais fácil de validar que String
  cor            String
  combustivel    Combustivel
  kmUltimaPass   Int?
  
  // RN33: Todo veículo pertence a UM cliente
  clienteId      Int
  cliente        Cliente   @relation(fields: [clienteId], references: [id], onDelete: Restrict)

  ordensServico  OrdemServico[]
}

enum Combustivel {
  GASOLINA
  DIESEL
  ETANOL
  FLEX
  GNV
  ELETRICO
  HIBRIDO
}

// ---------------------------------------------------------
// 3. O CATÁLOGO (SERVIÇOS E PEÇAS)
// ---------------------------------------------------------

// RN32: Categorias infinitas com Auto-relacionamento
model Categoria {
  id        Int         @id @default(autoincrement())
  nome      String
  paiId     Int?
  pai       Categoria?  @relation("HierarquiaCategoria", fields: [paiId], references: [id])
  filhas    Categoria[] @relation("HierarquiaCategoria")
  servicos  Servico[]
}

model Servico {
  id            Int       @id @default(autoincrement())
  codigo        String    @unique // RN04: Ex: MOT-014
  nome          String
  descricao     String    @db.Text
  precoTabela   Decimal   @db.Decimal(10, 2)
  tempoEstimado Int       // em minutos
  ativo         Boolean   @default(true)

  categoriaId   Int
  categoria     Categoria @relation(fields: [categoriaId], references: [id])

  itensOS       ItemOSServico[]
}

model Peca {
  id            Int       @id @default(autoincrement())
  codigo        String    @unique // RN04
  nome          String
  precoCusto    Decimal   @db.Decimal(10, 2)
  precoVenda    Decimal   @db.Decimal(10, 2)
  estoqueMinimo Int
  estoqueAtual  Int       @default(0)

  // RN19: Fornecedor opcional (pode ficar sem se ele sair da praça)
  fornecedorId  Int?
  fornecedor    Fornecedor? @relation(fields: [fornecedorId], references: [id], onDelete: SetNull)

  itensOS       ItemOSPeca[]
  movimentacoes MovimentacaoEstoque[]
}

model Fornecedor {
  id          Int     @id @default(autoincrement())
  razaoSocial String
  cnpj        String  @unique
  email       String?
  telefone    String?
  pecas       Peca[]
}

// ---------------------------------------------------------
// 4. A ORDEM DE SERVIÇO (O CORAÇÃO DO SISTEMA)
// ---------------------------------------------------------

model OrdemServico {
  id             Int       @id @default(autoincrement())
  numeroOS       String    @unique // RN05: Ex: OS-2026-000417
  abertura       DateTime  @default(now())
  previsaoEnt    DateTime?
  kmEntrada      Int
  observacoes    String?   @db.Text
  desconto       Decimal   @default(0) @db.Decimal(10, 2)
  etapa          Etapa     @default(ABERTA) // RN21

  unidadeId      Int
  unidade        Unidade   @relation(fields: [unidadeId], references: [id])
  veiculoId      Int
  veiculo        Veiculo   @relation(fields: [veiculoId], references: [id], onDelete: Restrict)

  // RN34: Sempre tem atendente. Mecânico é opcional.
  atendenteId    Int
  atendente      Colaborador @relation("AtendenteOS", fields: [atendenteId], references: [id])
  
  // RN18: Se mecânico for apagado, setar como Nulo (SetNull)
  mecanicoId     Int?
  mecanico       Colaborador? @relation("MecanicoOS", fields: [mecanicoId], references: [id], onDelete: SetNull)

  itensServico   ItemOSServico[]
  itensPeca      ItemOSPeca[]
  pagamentos     Pagamento[]
  movEstoque     MovimentacaoEstoque[]
}

enum Etapa {
  ABERTA
  ORCAMENTO
  APROVADA
  EM_EXECUCAO
  AGUARDANDO_PECA
  FINALIZADA
  ENTREGUE
  CANCELADA
}

// ---------------------------------------------------------
// 5. ITENS DA OS (RESOLVENDO A RN10)
// ---------------------------------------------------------

// Tabela intermediária explícita N:M
model ItemOSServico {
  id           Int          @id @default(autoincrement())
  quantidade   Int
  precoCobrado Decimal      @db.Decimal(10, 2) // OBRIGATÓRIO: congela o valor na OS

  osId         Int
  os           OrdemServico @relation(fields: [osId], references: [id], onDelete: Cascade) // RN16
  servicoId    Int
  servico      Servico      @relation(fields: [servicoId], references: [id], onDelete: Restrict)
  mecanicoId   Int
  mecanico     Colaborador  @relation(fields: [mecanicoId], references: [id])

  // RN07: O mesmo serviço não entra duas vezes na mesma OS
  @@unique([osId, servicoId])
}

model ItemOSPeca {
  id           Int          @id @default(autoincrement())
  quantidade   Int
  precoCobrado Decimal      @db.Decimal(10, 2) // OBRIGATÓRIO: congela o valor

  osId         Int
  os           OrdemServico @relation(fields: [osId], references: [id], onDelete: Cascade) // RN16
  pecaId       Int
  peca         Peca         @relation(fields: [pecaId], references: [id], onDelete: Restrict)

  // RN08: A mesma peça não entra duas vezes na mesma OS
  @@unique([osId, pecaId])
}

// ---------------------------------------------------------
// 6. PAGAMENTOS E ESTOQUE
// ---------------------------------------------------------

model Pagamento {
  id             Int          @id @default(autoincrement())
  valor          Decimal      @db.Decimal(10, 2)
  forma          FormaPagto
  numParcelas    Int          @default(1)
  dataHora       DateTime     @default(now())

  // O pagamento pertence à OS (Uma OS pode ter VÁRIOS pagamentos)
  osId           Int
  os             OrdemServico @relation(fields: [osId], references: [id], onDelete: Cascade)

  quemRecebeId   Int
  quemRecebe     Colaborador  @relation(fields: [quemRecebeId], references: [id])
}

enum FormaPagto {
  DINHEIRO
  PIX
  DEBITO
  CREDITO
  BOLETO
}

// RN13: Rastro de alteração de estoque
model MovimentacaoEstoque {
  id            Int          @id @default(autoincrement())
  tipo          TipoMovimento
  quantidade    Int
  motivo        String
  dataHora      DateTime     @default(now())

  pecaId        Int
  peca          Peca         @relation(fields: [pecaId], references: [id])
  
  responsavelId Int
  responsavel   Colaborador  @relation(fields: [responsavelId], references: [id])
  
  // A saída pode ou não estar ligada a uma OS
  osId          Int?
  os            OrdemServico? @relation(fields: [osId], references: [id])
}

enum TipoMovimento {
  ENTRADA
  SAIDA
}
